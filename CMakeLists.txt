cmake_minimum_required(VERSION 3.10)
project(RobotStriker CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)

# Semua source kecuali main.cpp masuk ke library inti (dipakai simulator & unit test)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/*.cpp)
list(FILTER CORE_SOURCES EXCLUDE REGEX ".*/main\\.cpp$")
add_library(striker_core STATIC ${CORE_SOURCES})
target_include_directories(striker_core PUBLIC ${CMAKE_SOURCE_DIR}/include)

if(EXISTS ${CMAKE_SOURCE_DIR}/src/main.cpp)
  add_executable(striker_sim src/main.cpp)
  target_link_libraries(striker_sim striker_core)
endif()

file(GLOB TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tests/*.cpp)
if(TEST_SOURCES)
  add_executable(run_tests ${TEST_SOURCES})
  target_link_libraries(run_tests striker_core)
  enable_testing()
  add_test(NAME unit_tests COMMAND run_tests)
endif()
